<!DOCTYPE html>
<html>
    <head>
        <meta charset="utf-8">
    </head>
    <body>
        <script>
            let n_ula = -1;

            do {
                n_ula = parseFloat(prompt("Podaj silnie"));
            } while (n_ula < 0);

            let silnia_ula = 1;

            if (n_ula != 0) {
                for (let i = 1; i <= n_ula; i++) {
                    silnia_ula *= i;
                }
            }

            document.write(`${n_ula}! = ${silnia_ula}`);
        </script>
    </body>
</html>
